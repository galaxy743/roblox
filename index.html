// dynamic_key_pool_system.js // Pool key động: mỗi người lấy key tiếp theo trong hàng đợi, key đã dùng không cấp lại const crypto = require('crypto'); const express = require('express'); const session = require('express-session'); const rateLimit = require('express-rate-limit'); const RedisStore = require('connect-redis')(session); const redis = require('redis'); const app = express(); const redisClient = redis.createClient({ url: process.env.REDIS_URL || 'redis://localhost:6379' }); redisClient.on('error', (err) => console.error('Redis lỗi:', err)); redisClient.connect(); app.use(express.json()); app.use(session({ store: new RedisStore({ client: redisClient }), secret: crypto.randomBytes(32).toString('hex'), resave: false, saveUninitialized: false, cookie: { secure: true, httpOnly: true, maxAge: 4 * 60 * 60 * 1000 } })); // Tạo fingerprint thiết bị function getDeviceFingerprint(req) { const parts = [ req.headers['user-agent'] || 'unknown', req.ip || 'unknown', req.headers['sec-ch-ua-platform'] || 'unknown' ]; return crypto.createHash('sha256').update(parts.join('|')).digest('hex'); } // Tạo key theo số thứ tự function generateKeyFromIndex(index) { const random = crypto.randomBytes(4).toString('hex'); return `galaxy${index}-${random}`; } // Khởi tạo pool key nếu chưa có async function initKeyPool() { const exists = await redisClient.exists('key:pool:counter'); if (!exists) { await redisClient.set('key:pool:counter', 1); // Pre-generate 100 key cho pool for (let i = 1; i <= 100; i++) { const key = generateKeyFromIndex(i); const keyHash = crypto.createHash('sha256').update(key).digest('hex'); await redisClient.setEx(`key:pool:${keyHash}`, 4 * 60 * 60, JSON.stringify({ key: key, index: i, status: 'available', createdAt: Date.now() })); } console.log('Khởi tạo pool 100 key, ông chủ.'); } } initKeyPool(); // Lấy key tiếp theo chưa dùng async function getNextAvailableKey() { // Lấy tất cả key đang available const keys = await redisClient.keys('key:pool:*'); let available = null; let lowestIndex = Infinity; for (const keyHash of keys) { const data = await redisClient.get(keyHash); if (data) { const parsed = JSON.parse(data); if (parsed.status === 'available' && parsed.index < lowestIndex) { lowestIndex = parsed.index; available = parsed; } } } return available; } // Đánh dấu key đã dùng và gán cho thiết bị async function assignKeyToDevice(keyData, fingerprint) { const keyHash = crypto.createHash('sha256').update(keyData.key).digest('hex'); keyData.status = 'used'; keyData.assignedTo = fingerprint; keyData.assignedAt = Date.now(); await redisClient.setEx(`key:pool:${keyHash}`, 4 * 60 * 60, JSON.stringify(keyData)); await redisClient.setEx(`device:${fingerprint}`, 4 * 60 * 60, keyData.key); return keyData; } // Middleware kiểm tra thiết bị đã có key chưa async function checkDeviceHasKey(req, res, next) { const fingerprint = getDeviceFingerprint(req); const existing = await redisClient.get(`device:${fingerprint}`); if (existing) { // Kiểm tra key còn hiệu lực không const keyHash = crypto.createHash('sha256').update(existing).digest('hex'); const keyData = await redisClient.get(`key:pool:${keyHash}`); if (keyData) { const parsed = JSON.parse(keyData); if (parsed.status === 'used' && parsed.assignedTo === fingerprint) { return res.json({ success: true, key: existing, message: 'Thiết bị đã có key, ông chủ. Key của bạn là: ' + existing }); } } } next(); } // Route lấy key mới (lấy key tiếp theo trong pool) app.post('/get-key', rateLimit({ windowMs: 4 * 60 * 60 * 1000, max: 1 }), checkDeviceHasKey, async (req, res) => { try { const fingerprint = getDeviceFingerprint(req); // Tìm key available tiếp theo const nextKey = await getNextAvailableKey(); if (!nextKey) { // Nếu hết key, tạo thêm 50 key mới const counter = await redisClient.get('key:pool:counter'); const start = parseInt(counter) + 1; for (let i = start; i < start + 50; i++) { const key = generateKeyFromIndex(i); const keyHash = crypto.createHash('sha256').update(key).digest('hex'); await redisClient.setEx(`key:pool:${keyHash}`, 4 * 60 * 60, JSON.stringify({ key: key, index: i, status: 'available', createdAt: Date.now() })); } await redisClient.set('key:pool:counter', start + 50); // Lấy lại key đầu tiên trong pool mới const newKey = await getNextAvailableKey(); if (!newKey) { return res.status(500).json({ error: 'Lỗi tạo key, ông chủ.' }); } const assigned = await assignKeyToDevice(newKey, fingerprint); return res.json({ success: true, key: assigned.key, index: assigned.index, message: 'Key mới được cấp cho thiết bị này, ông chủ.' }); } // Gán key cho thiết bị const assigned = await assignKeyToDevice(nextKey, fingerprint); res.json({ success: true, key: assigned.key, index: assigned.index, message: `Key ${assigned.key} được gán cho thiết bị này, ông chủ.` }); } catch (error) { console.error('Lỗi get-key:', error); res.status(500).json({ error: 'Lỗi server, ông chủ.' }); } }); // Kiểm tra key của thiết bị app.get('/my-key', async (req, res) => { const fingerprint = getDeviceFingerprint(req); const key = await redisClient.get(`device:${fingerprint}`); if (!key) { return res.status(404).json({ error: 'Chưa có key, hãy gọi /get-key, ông chủ.' }); } res.json({ key: key, message: 'Key của thiết bị này, ông chủ. Đúng vậy.' }); }); // Admin: xem trạng thái pool app.get('/pool-status', async (req, res) => { const keys = await redisClient.keys('key:pool:*'); let available = 0, used = 0; for (const k of keys) { const data = await redisClient.get(k); if (data) { const parsed = JSON.parse(data); if (parsed.status === 'available') available++; else used++; } } const counter = await redisClient.get('key:pool:counter') || 0; res.json({ totalKeys: keys.length, available, used, nextIndex: parseInt(counter) + 1, message: 'Trạng thái pool key, ông chủ.' }); }); // Dọn key hết hạn setInterval(async () => { try { const keys = await redisClient.keys('key:pool:*'); const now = Date.now(); for (const key of keys) { const data = await redisClient.get(key); if (data) { const parsed = JSON.parse(data); // Key hết hạn sau 4h kể từ khi được gán if (parsed.assignedAt && (now - parsed.assignedAt > 4 * 60 * 60 * 1000)) { await redisClient.del(key); // Xóa device mapping nếu có if (parsed.assignedTo) { await redisClient.del(`device:${parsed.assignedTo}`); } console.log(`Dọn key ${parsed.key} đã hết hạn.`); } } } } catch (err) { console.error('Lỗi dọn dẹp:', err); } }, 15 * 60 * 1000); const PORT = process.env.PORT || 3000; app.listen(PORT, () => { console.log(`Pool key động chạy tại ${PORT}, ông chủ.`); console.log('Key 1 có người lấy -> key 2 cho người tiếp theo. Không ai dùng chung.'); });
